1.Two Sum : Check if a pair with given sum exists in Array

Input: N = 5, arr[] = {2,6,5,8,11}, target = 14
Output : YES
Explanation: arr[1] + arr[3] = 14. So, the answer is “YES” for first variant for second variant output will be : [1,3].

Input: N = 5, arr[] = {2,6,5,8,11}, target = 15
Output : NO.
Explanation: There exist no such two numbers whose sum is equal to the target. 

//code(Brute solution)

class main{
    public static int[] akil(int[] arr,int target){
        int n=arr.length;
        for(int i=0;i<n;i++){
            for(int j=i+1;j<n;j++){
                if(arr[i]+arr[j]==target){
                    return new int[]{i,j};
                }
            }
        }
        return new int[]{-1,-1};
    }
    public static void main(String[] args){
        int[] arr = {2,6,5,8,11};
        int target = 14;
        
        main sol=new main();
        int[] a=sol.akil(arr,target);
        System.out.print(a[0]+","+a[1]);
    }
}

import java.util.Arrays;
class main{
    public int[] akil(int[] arr, int target){
        int n=arr.length;
        int[][] newarr=new int[n][2];
        for(int i=0;i<n;i++){
            newarr[i][0]=arr[i];
            newarr[i][1]=i;
        }
        Arrays.sort(newarr,(a,b)->Integer.compare(a[0],b[0]));
        
        int left=0,right=n-1;
        while(left<right){
            int sum=newarr[left][0]+newarr[right][0];
            if(sum==target){
                return new int[]{newarr[left][1],newarr[right][1]};
            }
            else if(sum<target) left++;
            else right--;
        }
        return new int[]{-1,-1};
    }
    public static void main(String[] args){
        int[] arr = {2, 6, 5, 8, 11};
        int target = 14;
        
        main sol=new main();
        int []a=sol.akil(arr,target);
        System.out.print(a[0]+","+a[1]);
    }
}

2.Sort an array of 0s, 1s and 2s

Problem Statement: Given an array nums consisting of only 0, 1, or 2. Sort the array in non-decreasing order. The sorting must be done in-place, without making a copy of the original array.

Examples
Input: nums = [1, 0, 2, 1, 0]
Output: [0, 0, 1, 1, 2]
Explanation: The nums array in sorted order has 2 zeroes, 2 ones and 1 two

Input: nums = [0, 0, 1, 1, 1]
Output: [0, 0, 1, 1, 1]

//code(optimal)
class main{
    public static void akil(int[] arr){
        int n=arr.length;
        int low=0,mid=0,high=n-1;
        while(mid<=high){
            if(arr[mid]==0){
                int temp=arr[low];
                arr[low]=arr[mid];
                arr[mid]=temp;
                low++;
                mid++;
            }
            else if(arr[mid]==1){
                mid++;
            }
            else{
                int temp=arr[high];
                arr[high]=arr[mid];
                arr[mid]=temp;
                high--;
                
            }
        }
    }
    public static void main(String[] args){
       int[]arr = {1, 0, 2, 1, 0};
       main sol=new main();
       sol.akil(arr);
       for(int a:arr){
           System.out.print(a);
       }
    }

4.Kadane's Algorithm : Maximum Subarray Sum in an Array
Example 1:
Input:
 nums = [2, 3, 5, -2, 7, -4]  
Output:
 15  
Explanation:
 The subarray from index 0 to index 4 has the largest sum = 15, which is the maximum sum of any contiguous subarray.

Example 2:
Input:
 nums = [-2, -3, -7, -2, -10, -4]  
Output:
 -2  
//code (brute force)
class main{
    public static int akil(int[] arr){
        int n=arr.length;
        int maxlen=Integer.MIN_VALUE;
        for(int i=0;i<n;i++){
            for(int j=i;j<n;j++){
                int sum=0;
                for(int k=i;k<=j;k++){
                    sum=sum+arr[k];
                }
                maxlen=Math.max(maxlen,sum);
            }
        }
        return maxlen;
    }
    public static void main(String[] args){
        int[] arr = { -2, 1, -3, 4, -1, 2, 1, -5, 4 };
        main sol=new main();
        int a=sol.akil(arr);
        System.out.print(a);
    }
}

5.Stock Buy And Sell
Input: prices = [7,1,5,3,6,4]
Output: 5
Explanation: Buy on day 2 (price = 1) and sell on day 5 (price = 6), profit = 6-1 = 5.
Note: That buying on day 2 and selling on day 1 is not allowed because you must buy before you sell.

Input: prices = [7,6,4,3,1]
Output: 0
Explanation: In this case, no transactions are done and the max profit = 0.

//code(brutal)
class main{
    public static int akil(int[] arr){
        int n=arr.length;
        int maxprofit=0;
        for(int i=0;i<n;i++){
            int profit=0;
            for(int j=i+1;j<n;j++){
                profit=arr[j]-arr[i];
                maxprofit=Math.max(maxprofit,profit);
                
            }
            
        }
        return maxprofit;
    }
    public static void main(String[] args){
        int[] arr = {7, 1, 5, 3, 6, 4};
        main sol=new main();
        int a=sol.akil(arr);
        System.out.print(a);
    }
}

class main{
    public static int akil(int[] arr){
        int n=arr.length;
        int maxprofit=0;
        int minprofit=Integer.MAX_VALUE;
        
        for(int i=0;i<n;i++){
            if(arr[i]<minprofit){
                minprofit=arr[i];
                
            }

            }
        }
        return maxprofit;

6.next_permutation : find next lexicographically greater permutation
Input: Arr[] = {1,3,2}
Output: {2,1,3}
Explanation: All permutations of {1,2,3} are {{1,2,3} , {1,3,2}, {2,13} , {2,3,1} , {3,1,2} , {3,2,1}}. So, the next permutation just after {1,3,2} is {2,1,3}.
Input : Arr[] = {3,2,1}
Output: {1,2,3}
Explanation : As we see all permutations of {1,2,3}, we find {3,2,1} at the last position. So, we have to return the lowest permutation.
//code(optimal)
class main{
    public static int[] akil(int[] arr){
        int n=arr.length;
        int index=-1;
        for(int i=n-2;i>=0;i--){
            if(arr[i]<arr[i+1]){
                index=i;
                break;
            }
        }
        if(index==-1){
            int start=0,end=n-1;
            while(start<end){
                int temp=arr[start];
                arr[start]=arr[end];
                arr[end]=temp;
                start++;
                end--;
            }
            return arr;
        }
        
        for(int i=n-1;i>=0;i--){
            if(arr[i]>arr[index]){
                int temp=arr[i];
                arr[i]=arr[index];
                arr[index]=temp;
                break;
            }
        }
        int start=index+1,end=n-1;
        while(start<end){
            int temp=arr[start];
            arr[start]=arr[end];
            arr[end]=temp;
            start++;
            end--;
        }
        return arr;
        }
    
    public static void main(String[] args){
        int[]arr={1,2,3};
        main sol=new main();
        int[] a=sol.akil(arr);
        for(int i:a){
            System.out.print(i);
        }
    }

8.Longest Consecutive Sequence in an Array
Example 1:
Input:
 nums = [100, 4, 200, 1, 3, 2]  
Output:
 4 

//code(brute)
class Main{

    public static boolean harish(int[] arr,int num){
        int n = arr.length;
        for(int i = 0; i < n; i++){
            if(arr[i] == num){
                return true;
            }
        }
        return false;
    }

    public static int akil(int[] arr){
        int n = arr.length;
        if(n == 0) return 0;

        int large = 1;

        for(int i = 0; i < n; i++){
            int x = arr[i];
            int count = 1;

            while(harish(arr, x + 1)){
                x = x + 1;
                count++;
            }

            large = Math.max(large, count); // ✅ inside loop
        }

        return large;
    }

    public static void main(String[] args){
        int[] arr = {100, 4, 200, 1, 3, 2};

        int a = akil(arr);
        System.out.print(a);
    }
}

9.Set Matrix Zero
Input: matrix=[[1,1,1],[1,0,1],[1,1,1]]
Output: [[1,0,1],[0,0,0],[1,0,1]]
Explanation: Since matrix[2][2]=0.Therfore the 2nd column and 2nd row wil be set to 0.

//code(brutal)
class main{
    public static void akil(int[][] arr){
        int m=arr.length;
        int n=arr[0].length;
        
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(arr[i][j]==0){
                    for(int col=0;col<n;col++){
                        if(arr[i][col]!=0){
                            arr[i][col]=-1;
                        }
                    }
                    for(int row=0;row<m;row++){
                        if(arr[row][j]!=0){
                        arr[row][j]=-1;
                    }
                }
                }
            }}
            
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(arr[i][j]==-1){
                    arr[i][j]=0;
                }
            }
        }
        }
            

    public static void main(String[] args){
        int[][]arr={{-1,1,1}};
        akil(arr);
        for(int[] i:arr){
            for(int j:i){
                System.out.print(j);
            }
            System.out.println();
        }
    }
}
