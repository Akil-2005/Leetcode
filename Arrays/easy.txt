2. Find Second Smallest and Second Largest Element in an array
//code

class main{
    public static void main(String[] args){
        int arr[]={1,2,3,4,5};
        int n=arr.length;
        int small=Integer.MAX_VALUE;
        int ssmall=Integer.MAX_VALUE;
 
        for(int i=0;i<n;i++){
            if(arr[i]<small){
                ssmall=small;
                small=arr[i];
            }
            else if(arr[i]<ssmall && arr[i]!=small){
                ssmall=arr[i];
 
            }
        }
 
        int large=Integer.MIN_VALUE;
        int slarge=Integer.MIN_VALUE;
 
        for(int i=0;i<n;i++){
            if(arr[i]>large ){
                slarge=large;
                large=arr[i];
            }
            else if(arr[i]>slarge && arr[i]!=slarge){
                slarge=arr[i];
            }
        }
        System.out.println(small);
        System.out.println(ssmall);
        System.out.println(large);
        System.out.println(slarge);
    }
}

4.Remove Duplicates in-place from Sorted Array
//code
class main{
 
    public static int akil(int[] arr, int n){
        int i=0;
        for(int j=1;j<n;j++){
            if(arr[i]!=arr[j]){
                i++;
                arr[i]=arr[j];
            }
        }
        return i+1;
    }
    public static void main(String[] args){
        int []arr={1,1,2,2,3,3,4,5};
        int n=arr.length;
 
        int k=akil(arr,n);
        for(int i=0;i<k;i++){
            System.out.print(arr[i]);
        }
 
    }
}

6.Rotate array by K elements
//code

class main{
    public static void rotate(int[] arr, int start, int end){
 
        while(start<end){
            int temp=arr[start];
 
            arr[start]=arr[end];
             arr[end]=temp;
             start++;
             end--;
        }
    }
    public static int[] reverse(int[] arr,int k,String direction){
        int n=arr.length;
        k=k&2;
        if(direction.equals("right")){
            rotate(arr,0,n-1);
            rotate(arr,0,k-1);
            rotate(arr,k,n-1);
        }
        return arr;
    }
    public static void main(String[] args){
        int[] arr={1,2,3,4,5,6,7};
        int k=2;	
        String direction="right";
 
        int[] a=reverse(arr,k,direction);
        for(int i:a){
            System.out.print(i);
        }
    }
}

class main{
    public static void zero(int[] arr){
        int n=arr.length;
        int j=-1;
        for(int i=0;i<n;i++){
            if(arr[i]==0){

                j=i;
                break;
            }
        }
        if(j==-1) return ;
 
        for(int i=j+1;i<n;i++){
            if(arr[i]!=0){
                int temp=arr[i];
                arr[i]=arr[j];
                arr[j]=temp;
                j++;
            }
        }
    }
 
    public static void main(String[] args){
        int[] arr={1,0,2,0,0,3};
        main a=new main();
        a.zero(arr);
        for(int i:arr){
            System.out.print(i);
        }
    }
}

11.Count Maximum Consecutive One's in the array
//code
class main{
    public static int akil(int[] arr){
        int n=arr.length;
        int count=0;
        int max_count=0;
        for(int i=0;i<n;i++){
            if(arr[i]==1){
                count++;
            }
            else{
                count=0;
            }
            max_count=Math.max(count,max_count);
        }
        return max_count;
    }
    public static void main(String[] args){
        int[] arr={1,1,1,1,1,0,0,0,1,1};
        main a=new main();
        int ans=a.akil(arr);
        System.out.print(ans);
    }
}

13.Longest Subarray with given Sum K(Positives)

class main{
    public static void akil(int arr[] ,int k){
        int n=arr.length;
        int maxlen=0;
        for(int i=0;i<n;i++){
            int count=0;
            for(int j=i;j<n;j++){
                count=count+arr[j];
            
            if(count==k){
                maxlen=Math.max(maxlen,j-i+1);
            }
            }
            
        }
        System.out.print(maxlen);
    }
    public static void main(String[] args){
         int[] arr = {10, 5, 2, 7, 1, 9};
         int k = 15 ;
         akil(arr,k);
    }
}

class main{
    public static int akil(int[] arr,int k){
        int n=arr.length;
        int right=0,left=0;
        int maxlen=0;
        int sum=arr[0];
        
        while(right<n){
            //3
            while(sum>k){
                sum=sum-arr[left];
                left++;
            }
            //2
            if(sum==k){
                maxlen=Math.max(maxlen,right-left+1);
            }
            //1
            right++;
            if(right<n){
                sum=sum+arr[right];
            }
        }
        return maxlen;
    }
    public static void main(String[] args){
         int[] arr = {10, 5, 2, 7, 1, 9};
        int k = 15;
        main sol=new main();
        int a=sol.akil(arr,k);
        System.out.print(a);
    }
}

14.Length of the longest subarray with zero Sum
//code(brutal solution)
import java.util.*;
class main{
    public static void akil(int[] arr){
        int n=arr.length;
        int maxlen=0;
        int sum=0;
        Map<Integer,Integer> sets=new HashMap<>();
        for(int i=0;i<n;i++){
            sum=sum+arr[i];
            if(sum==0){
                maxlen=i+1;
            }
            else if(sets.containsKey(sum)){
                maxlen=Math.max(maxlen,i-sets.get(sum));
            }
            else{
                sets.put(sum,i);
            }
        }
        System.out.print(maxlen);
    }
    public static void main(String[] args){
        int[] arr={9, -3, 3, -1, 6, -5};
        akil(arr);
    }
}
